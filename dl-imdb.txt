import numpy as np
import seaborn as sns
import matplotlib.pyplot as plt
from keras.datasets import imdb
from keras import models, layers, optimizers, losses, Input
from sklearn.metrics import classification_report, confusion_matrix, accuracy_score

from keras.models import Sequential
from keras.layers import Dense, Embedding, Dropout, BatchNormalization, Input, Flatten

from keras.preprocessing.sequence import pad_sequences

(x_train, y_train), (x_test, y_test) = imdb.load_data(num_words=10000)



xtrain = pad_sequences(x_train, maxlen=200)
xtest = pad_sequences(x_test, maxlen=200)

model = Sequential()
model.add(Input(shape=(200,)))
model.add(Embedding(input_dim=10000, output_dim=32))
model.add(Flatten())
model.add(Dense(128, activation='relu'))
model.add(BatchNormalization())
model.add(Dropout(0.5))
model.add(Dense(64, activation='relu'))
model.add(BatchNormalization())
model.add(Dropout(0.5))
model.add(Dense(16, activation='relu'))
model.add(Dense(1, activation='sigmoid'))

model.summary()

model.compile(optimizer='adam', loss='binary_crossentropy', metrics=['accuracy'])

history = model.fit(
    xtrain,
    y_train,
    epochs=5,
    batch_size=128,
    validation_split=0.2
)

loss, accuracy = model.evaluate(xtest, y_test)
pred= model.predict(xtest)

pred = np.where(pred > 0.5, 1, 0)
accuracy_score(y_test,pred)
cm = confusion_matrix(y_test, pred)

  

sns.heatmap(
    cm,
    annot=True,
    fmt='d',
    cmap='Blues',
    xticklabels=['Negative', 'Positive'],
    yticklabels=['Negative', 'Positive']
)
